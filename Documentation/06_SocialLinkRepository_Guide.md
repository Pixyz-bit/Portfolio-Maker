# 06 - SocialLink Repository Guide (Full CRUD Operations)

## 1. What is the `SocialLinkRepository`?
The **`SocialLinkRepository`** is the data access layer dedicated to managing external and social media link records in `dbo.SocialLinks`.
Because a user can have many social links (1-to-Many relationship), this repository provides full CRUD (Create, Read, Update, Delete) methods:
- **`Create`**: Inserts a new link (e.g. GitHub, LinkedIn) associated with a user's `UserID`.
- **`GetByUserId`**: Retrieves all social links belonging to a specific user as a `List<SocialLink>`.
- **`GetById`**: Retrieves a single link by its primary key `SocialLinkID` (useful when editing a specific link).
- **`Update`**: Modifies the platform name or URL of an existing link.
- **`Delete`**: Removes a link from the database.

---

## 2. SQL Schema Mapping
This repository operates on the **`dbo.SocialLinks`** table from `02_Alter_Profile_Affiliations_SocialLinks.sql`:

| Column | Data Type | Constraint | C# Model Property | Notes |
| :--- | :--- | :--- | :--- | :--- |
| `SocialLinkID` | `INT` | `IDENTITY(1,1)`, Primary Key | `link.SocialLinkID` | Auto-generated by SQL |
| `UserID` | `INT` | Foreign Key to `dbo.Users` | `link.UserID` | Links link to account |
| `SocialLinkName` | `NVARCHAR(50)` | `NOT NULL` | `link.SocialLinkName` | e.g. "GitHub", "LinkedIn" |
| `Link` | `NVARCHAR(500)` | `NOT NULL` | `link.Link` | e.g. "https://github.com/user" |
| `CreatedAt` | `DATETIME2` | `DEFAULT SYSUTCDATETIME()` | `link.CreatedAt` | Creation timestamp |

## 4. Entire Code Snippet for `SocialLinkRepository.cs`

File: `241611JalopPersonalWebsite/Repository/SocialLinkRepository.cs`

```csharp
using System;
using System.Collections.Generic;
using System.Data;
using System.Data.SqlClient;
using _241611JalopPersonalWebsite.Model;

namespace _241611JalopPersonalWebsite.Repository
{
    public static class SocialLinkRepository
    {
        // =========================================================================
        // 1. CREATE: Insert New SocialLink
        // =========================================================================
        public static bool Create(SocialLink link, out string errorMessage)
        {
            errorMessage = string.Empty;

            if (link == null)
            {
                errorMessage = "Social link details cannot be null.";
                return false;
            }

            if (link.UserID <= 0)
            {
                errorMessage = "A valid UserID is required.";
                return false;
            }

            if (string.IsNullOrWhiteSpace(link.SocialLinkName))
            {
                errorMessage = "Social link name is required (e.g., GitHub, LinkedIn).";
                return false;
            }

            if (string.IsNullOrWhiteSpace(link.Link))
            {
                errorMessage = "URL link is required.";
                return false;
            }

            try
            {
                using (SqlConnection conn = DatabaseConnection.GetConnection())
                {
                    conn.Open();

                    string query = @"
                        INSERT INTO dbo.SocialLinks (UserID, SocialLinkName, Link, CreatedAt)
                        VALUES (@UserID, @SocialLinkName, @Link, SYSUTCDATETIME());
                        SELECT SCOPE_IDENTITY();";

                    using (SqlCommand cmd = new SqlCommand(query, conn))
                    {
                        cmd.Parameters.Add("@UserID", SqlDbType.Int).Value = link.UserID;
                        cmd.Parameters.Add("@SocialLinkName", SqlDbType.NVarChar, 50).Value = link.SocialLinkName.Trim();
                        cmd.Parameters.Add("@Link", SqlDbType.NVarChar, 500).Value = link.Link.Trim();

                        object result = cmd.ExecuteScalar();
                        if (result != null && int.TryParse(result.ToString(), out int newId))
                        {
                            link.SocialLinkID = newId;
                            return true;
                        }
                        else
                        {
                            errorMessage = "Failed to retrieve generated SocialLinkID.";
                            return false;
                        }
                    }
                }
            }
            catch (Exception ex)
            {
                errorMessage = $"Database error: {ex.Message}";
                return false;
            }
        }

        // =========================================================================
        // 2. READ ALL: Get All SocialLinks by UserID
        // =========================================================================
        public static List<SocialLink> GetByUserId(int userId, out string errorMessage)
        {
            errorMessage = string.Empty;
            List<SocialLink> list = new List<SocialLink>();

            if (userId <= 0)
            {
                errorMessage = "Invalid UserID.";
                return list;
            }

            try
            {
                using (SqlConnection conn = DatabaseConnection.GetConnection())
                {
                    conn.Open();

                    string query = @"
                        SELECT SocialLinkID, UserID, SocialLinkName, Link, CreatedAt
                        FROM dbo.SocialLinks
                        WHERE UserID = @UserID
                        ORDER BY CreatedAt ASC";

                    using (SqlCommand cmd = new SqlCommand(query, conn))
                    {
                        cmd.Parameters.Add("@UserID", SqlDbType.Int).Value = userId;

                        using (SqlDataReader reader = cmd.ExecuteReader())
                        {
                            while (reader.Read())
                            {
                                list.Add(MapFromReader(reader));
                            }
                        }
                    }
                }

                return list;
            }
            catch (Exception ex)
            {
                errorMessage = $"Database error: {ex.Message}";
                return list;
            }
        }

        // =========================================================================
        // 3. READ ONE: Get Single SocialLink by SocialLinkID
        // =========================================================================
        public static SocialLink GetById(int socialLinkId, out string errorMessage)
        {
            errorMessage = string.Empty;

            if (socialLinkId <= 0)
            {
                errorMessage = "Invalid SocialLinkID.";
                return null;
            }

            try
            {
                using (SqlConnection conn = DatabaseConnection.GetConnection())
                {
                    conn.Open();

                    string query = @"
                        SELECT SocialLinkID, UserID, SocialLinkName, Link, CreatedAt
                        FROM dbo.SocialLinks
                        WHERE SocialLinkID = @SocialLinkID";

                    using (SqlCommand cmd = new SqlCommand(query, conn))
                    {
                        cmd.Parameters.Add("@SocialLinkID", SqlDbType.Int).Value = socialLinkId;

                        using (SqlDataReader reader = cmd.ExecuteReader())
                        {
                            if (reader.Read())
                            {
                                return MapFromReader(reader);
                            }
                            else
                            {
                                errorMessage = "Social link not found.";
                                return null;
                            }
                        }
                    }
                }
            }
            catch (Exception ex)
            {
                errorMessage = $"Database error: {ex.Message}";
                return null;
            }
        }

        // =========================================================================
        // 4. UPDATE: Update Existing SocialLink
        // =========================================================================
        public static bool Update(SocialLink link, out string errorMessage)
        {
            errorMessage = string.Empty;

            if (link == null)
            {
                errorMessage = "Social link details cannot be null.";
                return false;
            }

            if (link.SocialLinkID <= 0)
            {
                errorMessage = "Invalid SocialLinkID.";
                return false;
            }

            if (string.IsNullOrWhiteSpace(link.SocialLinkName))
            {
                errorMessage = "Social link name is required.";
                return false;
            }

            if (string.IsNullOrWhiteSpace(link.Link))
            {
                errorMessage = "URL link is required.";
                return false;
            }

            try
            {
                using (SqlConnection conn = DatabaseConnection.GetConnection())
                {
                    conn.Open();

                    string query = @"
                        UPDATE dbo.SocialLinks
                        SET SocialLinkName = @SocialLinkName,
                            Link = @Link
                        WHERE SocialLinkID = @SocialLinkID";

                    using (SqlCommand cmd = new SqlCommand(query, conn))
                    {
                        cmd.Parameters.Add("@SocialLinkID", SqlDbType.Int).Value = link.SocialLinkID;
                        cmd.Parameters.Add("@SocialLinkName", SqlDbType.NVarChar, 50).Value = link.SocialLinkName.Trim();
                        cmd.Parameters.Add("@Link", SqlDbType.NVarChar, 500).Value = link.Link.Trim();

                        int rowsAffected = cmd.ExecuteNonQuery();
                        if (rowsAffected > 0)
                        {
                            return true;
                        }
                        else
                        {
                            errorMessage = "Social link not found to update.";
                            return false;
                        }
                    }
                }
            }
            catch (Exception ex)
            {
                errorMessage = $"Database error: {ex.Message}";
                return false;
            }
        }

        // =========================================================================
        // 5. DELETE: Remove SocialLink by SocialLinkID
        // =========================================================================
        public static bool Delete(int socialLinkId, out string errorMessage)
        {
            errorMessage = string.Empty;

            if (socialLinkId <= 0)
            {
                errorMessage = "Invalid SocialLinkID.";
                return false;
            }

            try
            {
                using (SqlConnection conn = DatabaseConnection.GetConnection())
                {
                    conn.Open();

                    string query = "DELETE FROM dbo.SocialLinks WHERE SocialLinkID = @SocialLinkID";

                    using (SqlCommand cmd = new SqlCommand(query, conn))
                    {
                        cmd.Parameters.Add("@SocialLinkID", SqlDbType.Int).Value = socialLinkId;

                        int rowsAffected = cmd.ExecuteNonQuery();
                        if (rowsAffected > 0)
                        {
                            return true;
                        }
                        else
                        {
                            errorMessage = "Social link not found to delete.";
                            return false;
                        }
                    }
                }
            }
            catch (Exception ex)
            {
                errorMessage = $"Database error: {ex.Message}";
                return false;
            }
        }

        // =========================================================================
        // HELPER: Map SqlDataReader to SocialLink Model
        // =========================================================================
        private static SocialLink MapFromReader(SqlDataReader reader)
        {
            return new SocialLink
            {
                SocialLinkID = reader.GetInt32(reader.GetOrdinal("SocialLinkID")),
                UserID = reader.GetInt32(reader.GetOrdinal("UserID")),
                SocialLinkName = reader.GetString(reader.GetOrdinal("SocialLinkName")),
                Link = reader.GetString(reader.GetOrdinal("Link")),
                CreatedAt = reader.GetDateTime(reader.GetOrdinal("CreatedAt"))
            };
        }
    }
}
```

## 5. Instructions: How to Use Each Function in Code-Behind

### Instruction 1: How to Add a Link (`SocialLinkRepository.Create`)
```csharp
protected void btnAdd_Click(object sender, EventArgs e)
{
    int currentUserId = Convert.ToInt32(Session["UserID"]);

    // 1. Bundle into SocialLink model
    SocialLink newLink = new SocialLink
    {
        UserID = currentUserId,
        SocialLinkName = ddlPlatform.SelectedValue, // e.g. "GitHub"
        Link = txtUrl.Text.Trim()
    };

    // 2. Call Create
    if (SocialLinkRepository.Create(newLink, out string error))
    {
        lblMessage.ForeColor = System.Drawing.Color.Green;
        lblMessage.Text = "Social link added successfully!";
        LoadLinks(); // Refresh the list
    }
    else
    {
        lblMessage.ForeColor = System.Drawing.Color.Red;
        lblMessage.Text = error;
    }
}
```

---

### Instruction 2: How to Load & Bind Links (`SocialLinkRepository.GetByUserId`)
Call this in `Page_Load` to populate an ASP.NET `GridView` or `Repeater`:

```csharp
private void LoadLinks()
{
    int currentUserId = Convert.ToInt32(Session["UserID"]);

    // Fetch all links for the user
    List<SocialLink> links = SocialLinkRepository.GetByUserId(currentUserId, out string error);

    // Bind to GridView or Repeater
    gvSocialLinks.DataSource = links;
    gvSocialLinks.DataBind();
}
```

### Instruction 3: How to Edit & Update a Link (`SocialLinkRepository.GetById` and `Update`)
```csharp
// A. When the user clicks "Edit" on a specific row:
protected void gvSocialLinks_SelectedIndexChanged(object sender, EventArgs e)
{
    int selectedId = Convert.ToInt32(gvSocialLinks.SelectedDataKey.Value);

    // Fetch single link by ID
    SocialLink link = SocialLinkRepository.GetById(selectedId, out string error);
    if (link != null)
    {
        hfSocialLinkId.Value = link.SocialLinkID.ToString();
        ddlPlatform.SelectedValue = link.SocialLinkName;
        txtUrl.Text = link.Link;
        btnSave.Text = "Update Link";
    }
}

// B. When the user clicks "Save/Update":
protected void btnUpdate_Click(object sender, EventArgs e)
{
    SocialLink linkToUpdate = new SocialLink
    {
        SocialLinkID = Convert.ToInt32(hfSocialLinkId.Value),
        SocialLinkName = ddlPlatform.SelectedValue,
        Link = txtUrl.Text.Trim()
    };

    if (SocialLinkRepository.Update(linkToUpdate, out string error))
    {
        lblMessage.Text = "Link updated successfully!";
        LoadLinks();
    }
    else
    {
        lblMessage.Text = error;
    }
}
```
